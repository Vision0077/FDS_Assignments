1. Factorial and Fibonacci (recursive and non-recursive)

Run in Ubuntu: g++ p01.cpp -o p01 && ./p01

#include <bits/stdc++.h>
using namespace std;
long long factR(int n){ return n<=1 ? 1 : n*factR(n-1); }
long long factI(int n){ long long f=1; for(int i=2;i<=n;i++) f*=i; return f; }
long long fibR(int n){ return n<2 ? n : fibR(n-1)+fibR(n-2); }
void fibI(int n){
    long long a=0,b=1;
    for(int i=0;i<n;i++){ cout<<a<<" "; long long c=a+b; a=b; b=c; }
}
int main(){
    int n; cout<<"n: "; cin>>n;
    cout<<"Factorial (recursive): "<<factR(n)<<"\n";
    cout<<"Factorial (iterative): "<<factI(n)<<"\n";
    cout<<"Fibonacci (recursive): ";
    for(int i=0;i<n;i++) cout<<fibR(i)<<" ";
    cout<<"\nFibonacci (iterative): "; fibI(n); cout<<"\n";
}

#include <bits/stdc++.h>
using namespace std;
struct S{ int r; string n; float g; };
const int N=15;
S s[N]={{23,"Amit",8.5},{5,"Neha",9.1},{41,"Rahul",7.8},{17,"Sneha",9.1},{2,"Karan",8.2},
        {33,"Pooja",8.5},{9,"Rohit",7.2},{28,"Anita",9.4},{12,"Vikas",6.9},{36,"Priya",8.9},
        {4,"Sagar",8.2},{19,"Meera",9.6},{45,"Arjun",7.8},{8,"Kiran",8.0},{30,"Divya",9.0}};

void show(S a[],int n){ for(int i=0;i<n;i++) printf("%-4d %-8s %.1f\n",a[i].r,a[i].n.c_str(),a[i].g); }

void bubble(S a[],int n){                       // by roll no, ascending
    for(int i=0;i<n-1;i++)
        for(int j=0;j<n-1-i;j++)
            if(a[j].r>a[j+1].r) swap(a[j],a[j+1]);
}
void insertion(S a[],int n){                    // by name, A-Z
    for(int i=1;i<n;i++){
        S k=a[i]; int j=i-1;
        while(j>=0 && a[j].n>k.n){ a[j+1]=a[j]; j--; }
        a[j+1]=k;
    }
}
void quick(S a[],int l,int h){                  // by SGPA, descending
    if(l>=h) return;
    float p=a[h].g; int i=l-1;
    for(int j=l;j<h;j++) if(a[j].g>p) swap(a[++i],a[j]);
    swap(a[i+1],a[h]);
    quick(a,l,i); quick(a,i+2,h);
}
int main(){
    int c; S t[N];
    do{
        cout<<"\n1.Roll call(Bubble) 2.Alphabetical(Insertion) 3.Top 10(Quick) 4.Search SGPA 5.Search name(Binary) 0.Exit: ";
        cin>>c; copy(s,s+N,t);
        if(c==1){ bubble(t,N); show(t,N); }
        else if(c==2){ insertion(t,N); show(t,N); }
        else if(c==3){ quick(t,0,N-1); show(t,10); }
        else if(c==4){
            float g; cout<<"SGPA: "; cin>>g; bool f=0;
            for(int i=0;i<N;i++) if(fabs(t[i].g-g)<0.01){ show(t+i,1); f=1; }
            if(!f) cout<<"None\n";
        }
        else if(c==5){                          // binary search (iterative) on sorted names, prefix match
            string k; cout<<"Name/prefix: "; cin>>k; insertion(t,N);
            int lo=0, hi=N;
            while(lo<hi){ int m=(lo+hi)/2; if(t[m].n.compare(0,k.size(),k)<0) lo=m+1; else hi=m; }
            bool f=0;
            for(;lo<N && t[lo].n.compare(0,k.size(),k)==0;lo++){ show(t+lo,1); f=1; }
            if(!f) cout<<"Not found\n";
        }
    }while(c);

6. Queue using array: print job scheduling

Run in Ubuntu: g++ p06.cpp -o p06 && ./p06

#include <bits/stdc++.h>
using namespace std;
const int N=5;
string q[N]; int f=0, r=0, c=0;                 // front, rear, count (circular queue)
int main(){
    int ch; string j;
    do{
        cout<<"\n1.Add print job 2.Print next job 3.Show queue 0.Exit: "; cin>>ch;
        if(ch==1){
            if(c==N) cout<<"Queue full\n";
            else { cout<<"Job name: "; cin>>j; q[r]=j; r=(r+1)%N; c++; }
        }
        else if(ch==2){
            if(!c) cout<<"No jobs\n";
            else { cout<<"Printing "<<q[f]<<"\n"; f=(f+1)%N; c--; }
        }
        else if(ch==3){
            if(!c) cout<<"Empty";
            for(int i=0,k=f;i<c;i++,k=(k+1)%N) cout<<q[k]<<" ";
            cout<<"\n";
        }
    }while(ch);
}

9. Infix to prefix/postfix and evaluation

Run in Ubuntu: g++ p09.cpp -o p09 && ./p09

#include <bits/stdc++.h>
using namespace std;
int p(char c){ return c=='^'?3 : (c=='*'||c=='/')?2 : (c=='+'||c=='-')?1 : 0; }

string post(string s,bool strict){              // infix -> postfix (strict=1 used for prefix trick)
    stack<char> st; string o;
    for(char c:s){
        if(isalnum(c)) o+=c;
        else if(c=='(') st.push(c);
        else if(c==')'){ while(st.top()!='('){ o+=st.top(); st.pop(); } st.pop(); }
        else{
            while(!st.empty() && (strict ? p(st.top())>p(c) : p(st.top())>=p(c))){ o+=st.top(); st.pop(); }
            st.push(c);
        }
    }
    while(!st.empty()){ o+=st.top(); st.pop(); }
    return o;
}
string prefix(string s){                        // reverse, swap brackets, postfix, reverse
    reverse(s.begin(),s.end());
    for(char &c:s) if(c=='(') c=')'; else if(c==')') c='(';
    s=post(s,1); reverse(s.begin(),s.end());
    return s;
}
int ap(int a,int b,char o){ return o=='+'?a+b : o=='-'?a-b : o=='*'?a*b : o=='/'?a/b : (int)pow(a,b); }

int evalPost(string s){                         // left -> right
    stack<int> st;
    for(char c:s){
        if(isdigit(c)) st.push(c-'0');
        else{ int b=st.top(); st.pop(); int a=st.top(); st.pop(); st.push(ap(a,b,c)); }
    }
    return st.top();
}
int evalPre(string s){                          // right -> left (operands swap order)
    stack<int> st;
    for(int i=s.size()-1;i>=0;i--){
        char c=s[i];
        if(isdigit(c)) st.push(c-'0');
        else{ int a=st.top(); st.pop(); int b=st.top(); st.pop(); st.push(ap(a,b,c)); }
    }
    return st.top();
}
int main(){
    string e; cout<<"Infix (single-digit numbers, no spaces): "; cin>>e;
    string po=post(e,0), pr=prefix(e);
    cout<<"Postfix: "<<po<<"\nPrefix : "<<pr<<"\n";
    cout<<"Postfix value: "<<evalPost(po)<<"\nPrefix value : "<<evalPre(pr)<<"\n";
}

#include <bits/stdc++.h>
using namespace std;
struct N{ string s; N* nx; };
N* h=0;
void add(string s){                             // insert at end
    N* n=new N{s,0};
    if(!h) h=n; else { N* t=h; while(t->nx) t=t->nx; t->nx=n; }
}
void del(string s){
    N *t=h, *p=0;
    while(t && t->s!=s){ p=t; t=t->nx; }
    if(!t){ cout<<"Song not found\n"; return; }
    if(p) p->nx=t->nx; else h=t->nx;
    delete t;
}
void find(string s){
    int i=1;
    for(N* t=h;t;t=t->nx,i++) if(t->s==s){ cout<<"Found at position "<<i<<"\n"; return; }
    cout<<"Not found\n";
}
void rev(){ N *p=0,*c=h,*n; while(c){ n=c->nx; c->nx=p; p=c; c=n; } h=p; }
void show(){ int i=1; for(N* t=h;t;t=t->nx) cout<<i++<<". "<<t->s<<"\n"; if(!h) cout<<"Playlist empty\n"; }
int main(){

12. Circular doubly linked list: photo gallery

Run in Ubuntu: g++ p12.cpp -o p12

#include <bits/stdc++.h>
using namespace std;
struct N{ string s; N *nx, *pv; };
N *h=0, *cur=0;                                 // circular doubly linked list
void add(string s){
    N* n=new N{s,0,0};
    if(!h){ h=cur=n; n->nx=n->pv=n; }
    else { N* t=h->pv; t->nx=n; n->pv=t; n->nx=h; h->pv=n; }
}
int main(){
    int c; string s;
    do{
        cout<<"\n1.Add photo 2.Next 3.Previous 4.Show all 5.Slideshow(loop) 0.Exit: "; cin>>c;
        if(c==1){ cout<<"Photo name: "; cin>>s; add(s); }
        else if(c>=2 && c<=5 && !h) cout<<"Gallery empty\n";
        else if(c==2){ cur=cur->nx; cout<<"Now: "<<cur->s<<"\n"; }
        else if(c==3){ cur=cur->pv; cout<<"Now: "<<cur->s<<"\n"; }
        else if(c==4){ N* t=h; do{ cout<<t->s<<(t==cur?"  <-- current":"")<<"\n"; t=t->nx; }while(t!=h); }
        else if(c==5){ N* t=cur; for(int i=0;i<10;i++){ cout<<t->s<<" -> "; t=t->nx; } cout<<"...\n"; }
    }while(c);
}
